  <script>
    const PRIMIERA_SCORES = { 7: 21, 6: 18, 1: 16, 5: 15, 4: 14, 3: 13, 2: 12, 8: 10, 9: 10, 10: 10 };
    const SUITS = ['Denari', 'Coppe', 'Bastoni', 'Spade'];
    const ICONS = { Denari: '🟡', Coppe: '🏆', Bastoni: '🪵', Spade: '⚔️' };

    let deck = [], table = [], p1Hand = [], p2Hand = [];
    let p1Taken = [], p2Taken = [];
    let p1Scope = 0, p2Scope = 0;
    let turn = 1, lastTaker = 1, gameMode = 'single';

    function generateDeck() {
      deck = [];
      SUITS.forEach(suit => {
        for (let val = 1; val <= 10; val++) deck.push({ suit, val });
      });
      deck.sort(() => Math.random() - 0.5);
    }

    function initGame(mode) {
      gameMode = mode;
      document.getElementById('lobby').style.display = 'none';
      document.getElementById('modal-results').style.display = 'none';

      document.getElementById('p1-label').innerText = gameMode === 'bot_vs_bot' ? 'Bot 1' : 'Tu';
      document.getElementById('p2-label').innerText = gameMode === 'multi' ? 'Giocatore 2' : 'Bot 2';

      p1Scope = 0; p2Scope = 0;
      p1Taken = []; p2Taken = [];
      document.getElementById('p1-scope-count').innerText = '0';
      document.getElementById('p2-scope-count').innerText = '0';

      generateDeck();
      table = [deck.pop(), deck.pop(), deck.pop(), deck.pop()];
      dealNextHands();
      turn = 1;
      updateUI();

      if (gameMode === 'bot_vs_bot') setTimeout(executeBotTurn, 1000);
    }

    function dealNextHands() {
      if (deck.length === 0) return;
      p1Hand = [deck.pop(), deck.pop(), deck.pop()];
      p2Hand = [deck.pop(), deck.pop(), deck.pop()];
    }

    function buildCardElement(card, isHidden, clickHandler) {
      const el = document.createElement('div');
      if (isHidden) {
        el.className = 'card back';
      } else {
        el.className = `card ${card.suit}`;
        el.innerHTML = `
          <div class="val">${card.val}</div>
          <div class="suit-icon">${ICONS[card.suit]}</div>
          <div class="val" style="align-self: flex-end">${card.val}</div>
        `;
      }
      if (clickHandler) el.onclick = clickHandler;
      return el;
    }

    function updateUI() {
      const p1Box = document.getElementById('p1-hand');
      p1Box.innerHTML = '';
      p1Hand.forEach((card, i) => {
        const canPlay = turn === 1 && gameMode !== 'bot_vs_bot';
        p1Box.appendChild(buildCardElement(card, false, canPlay ? () => playHandCard(i) : null));
      });

      const p2Box = document.getElementById('p2-hand');
      p2Box.innerHTML = '';
      p2Hand.forEach((card, i) => {
        const isVisible = gameMode === 'multi' && turn === 2;
        const canPlay = turn === 2 && gameMode === 'multi';
        p2Box.appendChild(buildCardElement(card, !isVisible, canPlay ? () => playHandCard(i) : null));
      });

      const tableBox = document.getElementById('table-area');
      tableBox.innerHTML = '';
      table.forEach(card => tableBox.appendChild(buildCardElement(card, false)));
    }

    function playHandCard(index) {
      const activeHand = turn === 1 ? p1Hand : p2Hand;
      const activeTaken = turn === 1 ? p1Taken : p2Taken;
      const played = activeHand.splice(index, 1)[0];

      let matchIdx = table.findIndex(c => c.val === played.val);
      if (matchIdx !== -1) {
        activeTaken.push(played, table.splice(matchIdx, 1)[0]);
        lastTaker = turn;
        verifyScopa();
      } else {
        let combos = getSumCombinations(table, played.val);
        if (combos.length > 0) {
          let chosenCombo = combos[0];
          activeTaken.push(played);
          chosenCombo.forEach(c => {
            activeTaken.push(c);
            table = table.filter(item => item !== c);
          });
          lastTaker = turn;
          verifyScopa();
        } else {
          table.push(played);
        }
      }

      advanceGame();
    }

    function verifyScopa() {
      if (table.length === 0 && deck.length > 0) {
        if (turn === 1) { p1Scope++; document.getElementById('p1-scope-count').innerText = p1Scope; }
        else { p2Scope++; document.getElementById('p2-scope-count').innerText = p2Scope; }
      }
    }

    function getSumCombinations(arr, target) {
      let results = [];
      const search = (sub, remaining) => {
        let sum = sub.reduce((a, b) => a + b.val, 0);
        if (sum === target) results.push(sub);
        if (sum >= target) return;
        for (let i = 0; i < remaining.length; i++) {
          search([...sub, remaining[i]], remaining.slice(i + 1));
        }
      };
      search([], arr);
      return results;
    }

    function advanceGame() {
      if (p1Hand.length === 0 && p2Hand.length === 0) {
        if (deck.length > 0) dealNextHands();
        else { finishGame(); return; }
      }

      turn = turn === 1 ? 2 : 1;
      updateUI();

      if ((gameMode === 'single' && turn === 2) || gameMode === 'bot_vs_bot') {
        setTimeout(executeBotTurn, 800);
      }
    }

    function executeBotTurn() {
      const activeHand = turn === 1 ? p1Hand : p2Hand;
      if (activeHand.length === 0) return;

      let selectedIdx = 0;
      for (let i = 0; i < activeHand.length; i++) {
        if (table.some(c => c.val === activeHand[i].val)) {
          selectedIdx = i;
          break;
        }
      }
      playHandCard(selectedIdx);
    }

    function calcPrimiera(takenCards) {
      let total = 0;
      SUITS.forEach(suit => {
        let maxSuitScore = 0;
        takenCards.filter(c => c.suit === suit).forEach(c => {
          let score = PRIMIERA_SCORES[c.val] || 0;
          if (score > maxSuitScore) maxSuitScore = score;
        });
        total += maxSuitScore;
      });
      return total;
    }

    function finishGame() {
      if (lastTaker === 1) p1Taken.push(...table);
      else p2Taken.push(...table);
      table = [];

      let p1Pts = p1Scope, p2Pts = p2Scope;

      if (p1Taken.length > p2Taken.length) p1Pts++;
      else if (p2Taken.length > p1Taken.length) p2Pts++;

      let p1D = p1Taken.filter(c => c.suit === 'Denari').length;
      let p2D = p2Taken.filter(c => c.suit === 'Denari').length;
      if (p1D > p2D) p1Pts++;
      else if (p2D > p1D) p2Pts++;

      if (p1Taken.some(c => c.suit === 'Denari' && c.val === 7)) p1Pts++;
      if (p2Taken.some(c => c.suit === 'Denari' && c.val === 7)) p2Pts++;

      let p1P = calcPrimiera(p1Taken);
      let p2P = calcPrimiera(p2Taken);
      if (p1P > p2P) p1Pts++;
      else if (p2P > p1P) p2Pts++;

      document.getElementById('win-title').innerText = p1Pts > p2Pts ? "🏆 Vittoria P1!" : (p2Pts > p1Pts ? "🤖 Vittoria P2!" : "🤝 Pareggio!");
      
      document.getElementById('score-summary').innerHTML = `
        <div class="res-row"><span>Scope:</span><b>${p1Scope} - ${p2Scope}</b></div>
        <div class="res-row"><span>Carte (${p1Taken.length} vs ${p2Taken.length}):</span><b>${p1Taken.length > p2Taken.length ? '+1 P1' : (p2Taken.length > p1Taken.length ? '+1 P2' : '0')}</b></div>
        <div class="res-row"><span>Denari (${p1D} vs ${p2D}):</span><b>${p1D > p2D ? '+1 P1' : (p2D > p1D ? '+1 P2' : '0')}</b></div>
        <div class="res-row"><span>Primiera (${p1P} vs ${p2P}):</span><b>${p1P > p2P ? '+1 P1' : (p2P > p1P ? '+1 P2' : '0')}</b></div>
        <div class="res-row" style="margin-top:10px; font-size:1.2rem; color:var(--gold);"><span>PUNTI TOTALI:</span><b>${p1Pts} - ${p2Pts}</b></div>
      `;

      document.getElementById('modal-results').style.display = 'flex';
    }

    function closeResults() {
      document.getElementById('lobby').style.display = 'flex';
      document.getElementById('modal-results').style.display = 'none';
    }
  </script>
